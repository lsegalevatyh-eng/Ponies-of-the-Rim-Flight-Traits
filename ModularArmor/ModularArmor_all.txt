using System;
using System.Collections.Generic;
using RimWorld;
using UnityEngine;
using Verse;

namespace ModularArmor
{
    public class ArmorSystemDef : Def
    {
        public Type systemClass;
        public int order;

        // Used only to preserve the save layout of legacy built-in systems.
        public bool legacySaveRoot;

        public List<StatDef> affectedStats;

        public override IEnumerable<string> ConfigErrors()
        {
            foreach (string error in base.ConfigErrors())
                yield return error;

            if (systemClass == null ||
                !typeof(ArmorSystem).IsAssignableFrom(systemClass) ||
                systemClass.IsAbstract)
            {
                yield return "systemClass must be a concrete ArmorSystem.";
            }
            else if (systemClass.GetConstructor(Type.EmptyTypes) == null)
            {
                yield return
                    "systemClass needs a public parameterless constructor.";
            }
        }
    }

    public interface IArmorEnergy
    {
        float Energy { get; }
        float Capacity { get; }
        bool RunningPowerAvailable { get; }

        // Pays a one-time cost from stored energy.
        bool TrySpend(float amount);
    }

    public sealed class ArmorEnergyBudget
    {
        public float Available { get; private set; }
        public float Capacity { get; }

        public ArmorEnergyBudget(float available, float capacity)
        {
            Available = Mathf.Max(0f, available);
            Capacity = Mathf.Max(0f, capacity);
        }

        // Repairs, restarts and similar operations require full payment.
        // A failed payment does not consume any energy.
        public bool TrySpend(float amount)
        {
            if (float.IsNaN(amount) ||
                float.IsInfinity(amount) ||
                amount < 0f ||
                amount > Capacity + 0.000001f ||
                Available + 0.000001f < amount)
                return false;

            Available = Mathf.Max(0f, Available - amount);
            return true;
        }

        internal bool PayRunningCost(float amount)
        {
            amount = Mathf.Max(0f, amount);

            if (Available + 0.000001f < amount)
            {
                Available = 0f;
                return false;
            }

            Available = Mathf.Max(0f, Available - amount);
            return true;
        }
    }

    public abstract class ArmorSystem
    {
        public CompModuleSlots Host { get; private set; }
        public ArmorSystemDef Definition { get; private set; }

        internal void Bind(
            CompModuleSlots host,
            ArmorSystemDef definition)
        {
            Host = host;
            Definition = definition;
            Initialize();
        }

        public virtual void Initialize() { }
        public virtual void ExposeData() { }
        public virtual void AfterLoad() { }
        public virtual void Tick() { }
        public virtual void ModulesChanged() { }

        public virtual void ModuleInstalled(ThingDef module) { }
        public virtual void ModuleRemoved(ThingDef module) { }

        public virtual void ModuleToggled(
            ThingDef module,
            bool enabled) { }

        // Return null to allow the operation, or a reason to block it.
        public virtual string InstallBlockedReason(ThingDef module)
            => null;

        public virtual string RemoveBlockedReason(ThingDef module)
            => null;

        public virtual float CapacityContribution => 0f;
        public virtual bool IsEnergySource => false;
        public virtual float GenerationPer60 => 0f;
        public virtual float GenerateForTick() => 0f;

        public virtual void ModifyRunningCost(
            ThingDef module,
            ref float cost) { }

        // Priority payments, such as shield restarts.
        public virtual void BeforeRunningEnergy(
            ArmorEnergyBudget budget) { }

        // Payments performed after running costs, such as repairs.
        public virtual void AfterRunningEnergy(
            ArmorEnergyBudget budget) { }

        public virtual void PowerLost() { }
        public virtual void EnergyEmpty() { }

        public virtual float StatOffset(StatDef stat) => 0f;

        public virtual void BeforeDamage(
            ref DamageInfo damage,
            ref bool absorbed) { }

        public virtual bool AllowVerb(Verb verb) => true;
        public virtual void DrawWorn() { }

        public virtual IEnumerable<Gizmo> GetWornGizmos()
        {
            yield break;
        }

        public virtual IEnumerable<string> DescribeModule(

            yield break;
        }

        // Optional secondary bar for the energy panel.

    }

using System;
using System.Collections.Generic;
using System.Linq;
using RimWorld;
using UnityEngine;
using Verse;

namespace ModularArmor
{
    public partial class CompModuleSlots : ThingComp
    {
        private List<ThingDef> modules = new List<ThingDef>();
        private List<ThingDef> enabled = new List<ThingDef>();

        private List<ArmorSystem> systems;
        private Pawn previousWearer;
        private float previousMovementBonus = float.NaN;

        private CompProperties_ModuleSlots Props =>
            (CompProperties_ModuleSlots)props;

        public Pawn Wearer => (parent as Apparel)?.Wearer;
        public bool IsCuirass => Props.allowArmorReinforcement;
        public string ArmorLabel => parent.LabelCap.ToString();

        public Map CurrentMap =>
            Wearer != null && Wearer.Spawned
                ? Wearer.Map
                : parent.Spawned ? parent.Map : null;

        public IntVec3 CurrentPosition =>
            Wearer != null && Wearer.Spawned
                ? Wearer.Position
                : parent.Position;

        public IReadOnlyList<ThingDef> InstalledModules =>
            modules.AsReadOnly();

        public IReadOnlyList<ArmorSystem> Systems
        {
            get
            {
                EnsureSystems();
                return systems.AsReadOnly();
            }
        }

        public T GetSystem<T>() where T : class
        {
            EnsureSystems();
            return systems.OfType<T>().FirstOrDefault();
        }

        public IArmorEnergy EnergyService =>
            GetSystem<IArmorEnergy>();

        private void EnsureSystems()
        {
            if (systems != null)
                return;

            systems = new List<ArmorSystem>();

            foreach (ArmorSystemDef def in
                DefDatabase<ArmorSystemDef>.AllDefsListForReading
                    .OrderBy(d => d.order)
                    .ThenBy(d => d.defName))
            {
                if (def.systemClass == null ||
                    def.systemClass.IsAbstract ||
                    !typeof(ArmorSystem).IsAssignableFrom(def.systemClass))
                    continue;

                ArmorSystem system =
                    (ArmorSystem)Activator.CreateInstance(def.systemClass);

                systems.Add(system);
                system.Bind(this, def);
            }
        }

        public int SlotCount
        {
            get
            {
                QualityCategory quality;

                if (!parent.TryGetQuality(out quality))
                    quality = QualityCategory.Normal;

                return Props.SlotsFor(quality);
            }
        }

        public int InstalledCount =>
            modules.Sum(d => Math.Max(
                1,
                ModuleUtility.Info(d)?.slots ?? 1));

        public int FreeSlots =>
            Math.Max(0, SlotCount - InstalledCount);

        public string SlotStatus
        {
            get
            {
                string text = string.Format(
                    ModuleUtility.T("MA_Slots", "Slots: {0}/{1}"),
                    InstalledCount,
                    SlotCount);

                if (InstalledCount > SlotCount)
                {
                    text += "\n" + ModuleUtility.T(
                        "MA_SlotOverflow",
                        "Quality slot limit exceeded. Remove excess modules.");
                }

                return text;
            }
        }

        public bool IsEnabled(ThingDef module) =>
            enabled.Contains(module);

        public bool IsModuleEffective(ThingDef module)
        {
            ModuleExtension info = ModuleUtility.Info(module);

            if (info == null || !modules.Contains(module))
                return false;

            return !info.active ||
                (Wearer != null &&
                 EnergyService != null &&
                 EnergyService.Capacity > 0f &&
                 EnergyService.RunningPowerAvailable &&
                 IsEnabled(module));
        }

        public List<CompModuleSlots> GetWornComponents()
        {
            var result = new List<CompModuleSlots>();

            if (Wearer?.apparel != null)
            {
                foreach (Apparel apparel in Wearer.apparel.WornApparel)
                {
                    CompModuleSlots comp =
                        apparel.GetComp<CompModuleSlots>();

                    if (comp != null)
                        result.Add(comp);
                }
            }

            return result;
        }

        public float StatOffset(StatDef stat)
        {
            EnsureSystems();
            return systems.Sum(s => s.StatOffset(stat));
        }

        public float RunningConsumption
        {
            get
            {
                EnsureSystems();
                float result = 0f;

                foreach (ThingDef module in modules)
                {
                    ModuleExtension info = ModuleUtility.Info(module);

                    if (info == null ||
                        !info.active ||
                        !IsEnabled(module))
                        continue;

                    float cost = Mathf.Max(0f, info.consumption);

                    foreach (ArmorSystem system in systems)
                        system.ModifyRunningCost(module, ref cost);

                    result += Mathf.Max(0f, cost);
                }

                return result;
            }
        }

        public string ModuleDescription(ThingDef module)
        {
            ModuleExtension info = ModuleUtility.Info(module);

            if (info == null)
                return "";

            var lines = new List<string>
            {
                ModuleUtility.T("MA_SlotCost", "Slots: ") +
                Math.Max(1, info.slots),

                ModuleUtility.T("MA_InstallationTime", "Installation: ") +
                ModuleUtility.TimeText(info.installTicks)
            };

            if (info.active)
            {
                lines.Add(ModuleUtility.T(
                    "MA_Consumption",
                    "Base consumption / 60 ticks: ") +
                    info.consumption.ToString("0.##"));
            }
            else
            {
                lines.Add(ModuleUtility.T(
                    "MA_Passive",
                    "Passive — no energy required"));
            }

            foreach (ArmorSystem system in Systems)
                lines.AddRange(system.DescribeModule(module));

            return string.Join("\n", lines);
        }

        public bool CanInstall(ThingDef module)
        {
            ModuleExtension info = ModuleUtility.Info(module);

            if (parent.Destroyed || info == null)
                return false;

            // Identical modules are limited per armor piece.
            // Missing, zero or negative limits are treated as one.
            int maxCopies = Math.Max(1, info.maxCopies);

            if (modules.Count(d => d == module) >= maxCopies)
                return false;

            if (!(IsCuirass ? info.forCuirass : info.forHelmet))
                return false;

            if (InstalledCount + Math.Max(1, info.slots) > SlotCount)
                return false;

            // Identical copies may share their exclusive group.
            // Different module definitions in that group still conflict.
            if (!string.IsNullOrEmpty(info.exclusiveGroup) &&
                modules.Any(d =>
                    d != module &&
                    ModuleUtility.Info(d)?.exclusiveGroup ==
                    info.exclusiveGroup))
                return false;

            return Systems.All(s =>
                s.InstallBlockedReason(module) == null);
        }

        public void Toggle(ThingDef module)
        {
            ModuleExtension info = ModuleUtility.Info(module);

            if (info == null ||
                !info.active ||
                !modules.Contains(module))
                return;

            bool value = !IsEnabled(module);

            if (value && (EnergyService?.Capacity ?? 0f) <= 0f)
            {
                ModuleUtility.Reject(ModuleUtility.T(
                    "MA_NoLocalCapacity",
                    "No local energy storage"));
                return;
            }

            if (value)
                enabled.Add(module);
            else
                enabled.Remove(module);

            foreach (ArmorSystem system in Systems)
                system.ModuleToggled(module, value);

            NotifyStatsChanged();
        }

        private void NotifyModulesChanged()
        {
            foreach (ArmorSystem system in Systems)
                system.ModulesChanged();

            NotifyStatsChanged();
        }

        public void NotifyStatsChanged()
        {
            Pawn pawn = Wearer;

            foreach (StatDef stat in
                DefDatabase<StatDef>.AllDefsListForReading)
            {
                stat.Worker.ClearCacheForThing(parent);

                if (pawn != null)
                    stat.Worker.ClearCacheForThing(pawn);

                if (previousWearer != null && previousWearer != pawn)
                    stat.Worker.ClearCacheForThing(previousWearer);
            }

            previousWearer = pawn;
            previousMovementBonus = StatOffset(StatDefOf.MoveSpeed);
        }

        public override void CompTick()
        {
            base.CompTick();
            CheckInstallation();

            foreach (ArmorSystem system in Systems)
                system.Tick();

            float movement = StatOffset(StatDefOf.MoveSpeed);

            if (Wearer != previousWearer ||
                float.IsNaN(previousMovementBonus) ||
                !Mathf.Approximately(movement, previousMovementBonus))
                NotifyStatsChanged();
        }

        public override void PostExposeData()
        {
            base.PostExposeData();
            EnsureSystems();

            Scribe_Collections.Look(
                ref modules,
                "maInstalledModules",
                LookMode.Def);

            Scribe_Collections.Look(
                ref enabled,
                "maEnabledModules",
                LookMode.Def);

            ExposeInstallation();

            if (Scribe.mode == LoadSaveMode.PostLoadInit)
            {
                modules = modules ?? new List<ThingDef>();
                enabled = enabled ?? new List<ThingDef>();

                modules.RemoveAll(d => d == null);
                enabled.RemoveAll(d => d == null || !modules.Contains(d));
            }

            foreach (ArmorSystem system in systems)
            {
                if (system.Definition.legacySaveRoot)
                {
                    system.ExposeData();
                }
                else
                {
                    Scribe.EnterNode(
                        "maSystem_" + system.Definition.defName);

                    try
                    {
                        system.ExposeData();
                    }
                    finally
                    {
                        Scribe.ExitNode();
                    }
                }
            }

            if (Scribe.mode == LoadSaveMode.PostLoadInit)
            {
                foreach (ArmorSystem system in systems)
                    system.AfterLoad();

                previousWearer = null;
                previousMovementBonus = float.NaN;
            }
        }

        public override void PostPreApplyDamage(
            ref DamageInfo damage,
            out bool absorbed)
        {
            absorbed = false;

            foreach (ArmorSystem system in Systems)
            {
                system.BeforeDamage(ref damage, ref absorbed);

                if (absorbed)
                    break;
            }
        }

        public override bool CompAllowVerbCast(Verb verb)
        {
            return Systems.All(s => s.AllowVerb(verb));
        }

        public override void CompDrawWornExtras()
        {
            base.CompDrawWornExtras();

            foreach (ArmorSystem system in Systems)
                system.DrawWorn();
        }

        public override string CompInspectStringExtra()
        {
            return SlotStatus +
                (InstallationPending ? "\n" + InstallationStatus : "");
        }

        private Command_Action SettingsCommand()
        {
            return new Command_Action
            {
                defaultLabel = ModuleUtility.T(
                    "MA_Settings",
                    "Armor settings"),

                defaultDesc = ArmorLabel + "\n\n" + SlotStatus,
                icon = parent.def.uiIcon,

                action = () => Find.WindowStack.Add(
                    new Dialog_ArmorSettings(this))
            };
        }

        public override IEnumerable<Gizmo> CompGetGizmosExtra()
        {
            if (parent.Spawned)
                yield return SettingsCommand();
        }

        public override IEnumerable<Gizmo> CompGetWornGizmosExtra()
        {
            if (Wearer == null ||
                Wearer.Faction != Faction.OfPlayer)
                yield break;

            yield return SettingsCommand();

            foreach (ArmorSystem system in Systems)
            {
                foreach (Gizmo gizmo in system.GetWornGizmos())
                    yield return gizmo;
            }
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using RimWorld;
using UnityEngine;
using Verse;
using Verse.AI;

namespace ModularArmor
{
    public partial class CompModuleSlots
    {
        private Pawn installingPawn;
        private ThingDef installingDef;
        private int installationProgressTicks;
        private bool removingModule;

        // The name is preserved for compatibility with the existing UI.
        // This now represents either installation or removal.
        public bool InstallationPending =>
            installingPawn != null && installingDef != null;

        public bool RemovalPending =>
            InstallationPending && removingModule;

        public ThingDef PendingModuleDef =>
            InstallationPending ? installingDef : null;

        private int CurrentOperationDuration
        {
            get
            {
                ModuleExtension info =
                    ModuleUtility.Info(installingDef);

                return Math.Max(
                    1,
                    removingModule
                        ? info?.removeTicks ?? 2500
                        : info?.installTicks ?? 2500);
            }
        }

        public float InstallationFraction =>
            !InstallationPending ? 0f :
            Mathf.Clamp01(
                installationProgressTicks /
                (float)CurrentOperationDuration);

        public string InstallationStatus =>
            !InstallationPending ? "" :
            (removingModule
                ? ModuleUtility.T("MA_Removing", "Removing: ")
                : ModuleUtility.T("MA_Installing", "Installing: ")) +
            installingDef.LabelCap + " — " +
            (InstallationFraction * 100f).ToString("0") + "%";

        private void ExposeInstallation()
        {
            Scribe_References.Look(
                ref installingPawn,
                "maInstallingPawn");

            Scribe_Defs.Look(
                ref installingDef,
                "maInstallingDef");

            Scribe_Values.Look(
                ref installationProgressTicks,
                "maInstallationProgress",
                0);

            Scribe_Values.Look(
                ref removingModule,
                "maRemovingModule",
                false);

            if (Scribe.mode == LoadSaveMode.PostLoadInit &&
                !InstallationPending)
                ClearInstallation();
        }

        public static bool InInventory(Pawn pawn, Thing thing)
        {
            return pawn?.inventory != null &&
                pawn.inventory.innerContainer.Contains(thing);
        }

        public bool IsAvailable(
            Thing thing,
            Pawn inventoryOwner)
        {
            if (thing == null ||
                thing.Destroyed ||
                thing.stackCount <= 0)
                return false;

            if (InInventory(inventoryOwner, thing))
                return true;

            return CurrentMap != null &&
                thing.Spawned &&
                thing.Map == CurrentMap &&
                !thing.IsForbidden(Faction.OfPlayer) &&
                thing.Position.DistanceToSquared(CurrentPosition) <= 4;
        }

        public List<Thing> NearbyThings(
            Func<ThingDef, bool> filter,
            Pawn inventoryOwner)
        {
            var result = new List<Thing>();

            if (inventoryOwner?.inventory != null)
            {
                foreach (Thing thing in
                    inventoryOwner.inventory.innerContainer)
                {
                    if (filter(thing.def) &&
                        IsAvailable(thing, inventoryOwner))
                        result.Add(thing);
                }
            }

            if (CurrentMap != null)
            {
                foreach (ThingDef def in
                    DefDatabase<ThingDef>.AllDefsListForReading)
                {
                    if (!filter(def))
                        continue;

                    foreach (Thing thing in
                        CurrentMap.listerThings.ThingsOfDef(def))
                    {
                        if (IsAvailable(thing, inventoryOwner))
                            result.Add(thing);
                    }
                }
            }

            return result;
        }

        public bool ReturnItem(Thing item)
        {
            if (Wearer?.inventory != null &&
                Wearer.inventory.innerContainer.TryAdd(item))
                return true;

            return CurrentMap != null &&
                GenPlace.TryPlaceThing(
                    item,
                    CurrentPosition,
                    CurrentMap,
                    ThingPlaceMode.Near);
        }

        private static string BusyOperationMessage =>
            ModuleUtility.T(
                "MA_MaintenanceBusy",
                "Finish installation or removal, or cancel the pawn's job.");

        public void OpenInstallMenu()
        {
            if (InstallationPending)
            {
                ModuleUtility.Reject(BusyOperationMessage);
                return;
            }

            if (FreeSlots <= 0)
            {
                ModuleUtility.Reject(ModuleUtility.T(
                    "MA_NoFreeSlots",
                    "No free module slots."));
                return;
            }

            if (Wearer != null)
            {
                OpenModuleMenu(Wearer);
                return;
            }

            var options = new List<FloatMenuOption>();

            if (parent.Spawned)
            {
                foreach (Pawn pawn in
                    parent.Map.mapPawns.FreeColonistsSpawned)
                {
                    if (pawn.Dead ||
                        pawn.Downed ||
                        !pawn.CanReach(
                            parent,
                            PathEndMode.Touch,
                            Danger.Deadly))
                        continue;

                    Pawn selected = pawn;

                    options.Add(new FloatMenuOption(
                        selected.LabelShortCap.ToString(),
                        () => OpenModuleMenu(selected)));
                }
            }

            if (options.Count == 0)
            {
                options.Add(new FloatMenuOption(
                    ModuleUtility.T(
                        "MA_NoInstaller",
                        "No available installer"),
                    null));
            }

            Find.WindowStack.Add(new FloatMenu(options));
        }

        private void OpenModuleMenu(Pawn pawn)
        {
            if (InstallationPending)
            {
                ModuleUtility.Reject(BusyOperationMessage);
                return;
            }

            var options = new List<FloatMenuOption>();

            foreach (var group in NearbyThings(CanInstall, pawn)
                .GroupBy(t => t.def)
                .OrderBy(g => g.Key.label))
            {
                Thing selected = group.First();

                options.Add(new FloatMenuOption(
                    selected.def.LabelCap + " — " +
                    ModuleUtility.TimeText(
                        ModuleUtility.Info(selected.def).installTicks),
                    () => StartInstallation(pawn, selected)));
            }

            if (options.Count == 0)
            {
                options.Add(new FloatMenuOption(
                    ModuleUtility.T(
                        "MA_NoAvailableModules",
                        "No compatible modules nearby or in inventory."),
                    null));
            }

            Find.WindowStack.Add(new FloatMenu(options));
        }

        private void StartInstallation(Pawn pawn, Thing module)
        {
            if (InstallationPending ||
                pawn == null ||
                !pawn.Spawned ||
                pawn.Dead ||
                pawn.Downed ||
                !IsAvailable(module, pawn) ||
                !CanInstall(module.def))
                return;

            JobDef jobDef =
                DefDatabase<JobDef>.GetNamed("MA_InstallModule");

            installingPawn = pawn;
            installingDef = module.def;
            installationProgressTicks = 0;
            removingModule = false;

            Job job = JobMaker.MakeJob(
                jobDef,
                parent,
                module,
                pawn);

            pawn.jobs.TryTakeOrderedJob(job, JobTag.Misc);
        }

        private void CheckInstallation()
        {
            if (!InstallationPending)
                return;

            Job job = installingPawn.CurJob;

            string expectedJob = removingModule
                ? "MA_RemoveModule"
                : "MA_InstallModule";

            if (installingPawn.Dead ||
                job == null ||
                job.def.defName != expectedJob ||
                job.targetA.Thing != parent)
                ClearInstallation();
        }

        private void ClearInstallation()
        {
            installingPawn = null;
            installingDef = null;
            installationProgressTicks = 0;
            removingModule = false;
        }

        public void SetInstallationProgress(
            Pawn pawn,
            int elapsed)
        {
            if (installingPawn == pawn)
                installationProgressTicks = Math.Max(0, elapsed);
        }

        public bool FinishInstallation(
            Pawn pawn,
            Thing module)
        {
            if (removingModule ||
                module == null ||
                module.Destroyed ||
                installingPawn != pawn ||
                module.def != installingDef ||
                !CanInstall(module.def))
                return false;

            Thing consumed = module.SplitOff(1);
            ThingDef def = consumed.def;

            modules.Add(def);
            consumed.Destroy(DestroyMode.Vanish);

            ClearInstallation();

            foreach (ArmorSystem system in Systems)
                system.ModuleInstalled(def);

            NotifyModulesChanged();
            return true;
        }

        // Checks removal restrictions without checking operation ownership.
        // Used both by the menu and by the running removal job.
        public string RemovalBlockedReason(ThingDef module)
        {
            if (parent.Destroyed ||
                module == null ||
                !modules.Contains(module))
            {
                return ModuleUtility.T(
                    "MA_ModuleNotInstalled",
                    "Module is not installed.");
            }

            foreach (ArmorSystem system in Systems)
            {
                string reason =
                    system.RemoveBlockedReason(module);

                if (reason != null)
                    return reason;
            }

            return null;
        }

        // Starts a removal job instead of removing the module immediately.
        public void RemoveModule(ThingDef module)
        {
            if (InstallationPending)
            {
                ModuleUtility.Reject(BusyOperationMessage);
                return;
            }

            string reason = RemovalBlockedReason(module);

            if (reason != null)
            {
                ModuleUtility.Reject(reason);
                return;
            }

            if (Wearer != null)
            {
                StartRemoval(Wearer, module);
                return;
            }

            var options = new List<FloatMenuOption>();

            if (parent.Spawned)
            {
                foreach (Pawn pawn in
                    parent.Map.mapPawns.FreeColonistsSpawned)
                {
                    if (pawn.Dead ||
                        pawn.Downed ||
                        !pawn.CanReach(
                            parent,
                            PathEndMode.Touch,
                            Danger.Deadly))
                        continue;

                    Pawn selected = pawn;

                    options.Add(new FloatMenuOption(
                        selected.LabelShortCap + " — " +
                        ModuleUtility.TimeText(
                            ModuleUtility.Info(module)?.removeTicks ?? 2500),
                        () => StartRemoval(selected, module)));
                }
            }

            if (options.Count == 0)
            {
                options.Add(new FloatMenuOption(
                    ModuleUtility.T(
                        "MA_NoInstaller",
                        "No available installer"),
                    null));
            }

            Find.WindowStack.Add(new FloatMenu(options));
        }

        private void StartRemoval(
            Pawn pawn,
            ThingDef module)
        {
            if (InstallationPending)
            {
                ModuleUtility.Reject(BusyOperationMessage);
                return;
            }

            if (pawn == null ||
                !pawn.Spawned ||
                pawn.Dead ||
                pawn.Downed)
            {
                ModuleUtility.Reject(ModuleUtility.T(
                    "MA_WearerUnavailable",
                    "Wearer is unavailable"));
                return;
            }

            string reason = RemovalBlockedReason(module);

            if (reason != null)
            {
                ModuleUtility.Reject(reason);
                return;
            }

            if (Wearer != null && Wearer != pawn)
                return;

            if (Wearer == null &&
                (!parent.Spawned ||
                 !pawn.CanReach(
                     parent,
                     PathEndMode.Touch,
                     Danger.Deadly)))
                return;

            JobDef jobDef =
                DefDatabase<JobDef>.GetNamed("MA_RemoveModule");

            installingPawn = pawn;
            installingDef = module;
            installationProgressTicks = 0;
            removingModule = true;

            Job job = JobMaker.MakeJob(jobDef, parent);
            job.SetTarget(TargetIndex.C, pawn);

            pawn.jobs.TryTakeOrderedJob(job, JobTag.Misc);
        }

        public bool IsRemovalInProgressFor(
            Pawn pawn,
            ThingDef module)
        {
            return RemovalPending &&
                installingPawn == pawn &&
                installingDef == module;
        }

        public bool FinishRemoval(
            Pawn pawn,
            ThingDef module)
        {
            if (!IsRemovalInProgressFor(pawn, module))
                return false;

            string reason = RemovalBlockedReason(module);

            if (reason != null)
            {
                ModuleUtility.Reject(reason);
                return false;
            }

            Thing item = ThingMaker.MakeThing(module);

            // Remove the installed module only after its item
            // has been returned to inventory or placed on the ground.
            if (!ReturnItem(item))
            {
                item.Destroy(DestroyMode.Vanish);

                ModuleUtility.Reject(ModuleUtility.T(
                    "MA_CannotRemove",
                    "Cannot return the module to inventory or place it on the ground."));

                return false;
            }

            modules.Remove(module);
            enabled.Remove(module);

            ClearInstallation();

            foreach (ArmorSystem system in Systems)
                system.ModuleRemoved(module);

            NotifyModulesChanged();
            return true;
        }
    }

            return def?.GetModExtension<ModuleExtension>();
        }

        public static string T(string key, string englishFallback)
        {
            return key.CanTranslate()
                ? key.Translate().ToString()
                : englishFallback;
        }

        // Compatibility overload for extensions using the previous API.
        // The old language-specific fallback is intentionally ignored.
        public static string T(
            string key,
            string unusedFallback,
            string englishFallback)
        {
            return T(key, englishFallback);
        }

        public static string Bonus(StatDef stat, float value)
        {
            string sign = value >= 0f ? "+" : "";

            if (stat == StatDefOf.ArmorRating_Sharp ||
                stat == StatDefOf.ArmorRating_Blunt ||
                stat == StatDefOf.ArmorRating_Heat)
            {
                return sign + (value * 100f).ToString("0.#") +
                    T("MA_PP", " percentage points");
            }

            string suffix =
                stat == StatDefOf.Insulation_Cold ||
                stat == StatDefOf.Insulation_Heat
                    ? " °C"
                    : "";

            return sign + value.ToString("0.##") + suffix;
        }

        public static string TimeText(int ticks)
        {
            return (Math.Max(0, ticks) / 2500f).ToString("0.00") +
                T("MA_GameHoursShort", " h");
        }

        public static void Reject(string text)
        {
            Messages.Message(
                text,
                MessageTypeDefOf.RejectInput,
                false);
        }
    }

using System;
using System.Collections.Generic;
using RimWorld;
using UnityEngine;
using Verse;
using Verse.AI;

namespace ModularArmor
{
    public class JobDriver_InstallModule : JobDriver
    {
        private Thing Armor =>
            job.GetTarget(TargetIndex.A).Thing;

        private Thing Module =>
            job.GetTarget(TargetIndex.B).Thing;

        private CompModuleSlots Comp =>
            (Armor as ThingWithComps)?.GetComp<CompModuleSlots>();

        private bool ModuleInInventory =>
            pawn.inventory != null &&
            pawn.inventory.innerContainer.Contains(Module);

        public override bool TryMakePreToilReservations(
            bool errorOnFailed)
        {
            if (Armor == null ||
                Armor.Destroyed ||
                Module == null ||
                Module.Destroyed ||
                Comp == null)
                return false;

            if (Comp.Wearer != null && Comp.Wearer != pawn)
                return false;

            if (Armor.Spawned &&
                !pawn.Reserve(
                    Armor,
                    job,
                    1,
                    -1,
                    null,
                    errorOnFailed))
                return false;

            if (Module.Spawned &&
                !pawn.Reserve(
                    Module,
                    job,
                    1,
                    -1,
                    null,
                    errorOnFailed))
                return false;

            return Module.Spawned || ModuleInInventory;
        }

        protected override IEnumerable<Toil> MakeNewToils()
        {
            this.FailOn(() =>
                Armor == null ||
                Armor.Destroyed ||
                Module == null ||
                Module.Destroyed ||
                Comp == null ||
                !Comp.CanInstall(Module.def) ||
                (Comp.Wearer != null && Comp.Wearer != pawn) ||
                (Comp.Wearer == null && !Armor.Spawned));

            int duration = Math.Max(
                1,
                ModuleUtility.Info(Module.def)?.installTicks ?? 2500);

            Toil acquire = new Toil
            {
                defaultCompleteMode = ToilCompleteMode.Instant
            };

            acquire.initAction = () =>
            {
                if (pawn.inventory == null)
                {
                    EndJobWith(JobCondition.Incompletable);
                    return;
                }

                if (ModuleInInventory && Module.stackCount == 1)
                    return;

                Thing item = Module.SplitOff(1);

                if (!pawn.inventory.innerContainer.TryAdd(item, false))
                {
                    if (pawn.Spawned)
                    {
                        GenPlace.TryPlaceThing(
                            item,
                            pawn.Position,
                            pawn.Map,
                            ThingPlaceMode.Near);
                    }

                    EndJobWith(JobCondition.Incompletable);
                    return;
                }

                // An interrupted installation leaves the item
                // in the installer's inventory.
                job.SetTarget(TargetIndex.B, item);
            };

            Toil wait = Toils_General.Wait(duration);
            Action originalTickAction = wait.tickAction;

            wait.tickAction = () =>
            {
                originalTickAction?.Invoke();

                Comp?.SetInstallationProgress(
                    pawn,
                    Math.Max(0, duration - ticksLeftThisToil));
            };

            // Target C is the installer.
            // Display the standard progress bar above the pawn.
            wait.WithProgressBar(
                TargetIndex.C,
                () => Mathf.Clamp01(
                    1f - ticksLeftThisToil / (float)duration));

            // Inventory items do not need to be collected from the ground.
            yield return Toils_Jump.JumpIf(
                acquire,
                () => ModuleInInventory);

            yield return Toils_Goto.GotoThing(
                TargetIndex.B,
                PathEndMode.Touch);

            yield return acquire;

            // Worn armor can be serviced without moving.
            yield return Toils_Jump.JumpIf(
                wait,
                () => Comp.Wearer == pawn);

            // Approach armor lying on the ground.
            yield return Toils_Goto.GotoThing(
                TargetIndex.A,
                PathEndMode.Touch);

            yield return wait;

            Toil finish = new Toil
            {
                defaultCompleteMode = ToilCompleteMode.Instant
            };

            finish.initAction = () =>
            {
                if (!Comp.FinishInstallation(pawn, Module))
                    EndJobWith(JobCondition.Incompletable);
            };

            yield return finish;
        }
    }
}

        public int durationTicks = 10000;

        // Energy paid once per consumption interval.

    public class ArmorCloakingSystem : ArmorSystem
    {
        private int activeUntil;
        private int cooldownUntil;
        private int nextEnergyPaymentTick;

        private Hediff effect;
        private Pawn effectPawn;
        private ThingDef activeModule;

        private int Now => Find.TickManager.TicksGame;

        private ThingDef Module =>
            Host.InstalledModules.FirstOrDefault(
                d => d.GetModExtension<CloakingModuleExtension>() != null);

        private CloakingModuleExtension Settings =>
            Module?.GetModExtension<CloakingModuleExtension>();

        private bool Active => effect != null;

        public override void ExposeData()
        {
            Scribe_Values.Look(ref activeUntil, "activeUntil", 0);
            Scribe_Values.Look(ref cooldownUntil, "cooldownUntil", 0);

            Scribe_Values.Look(
                ref nextEnergyPaymentTick,
                "nextEnergyPaymentTick",
                0);

            Scribe_References.Look(ref effect, "effect");
            Scribe_References.Look(ref effectPawn, "effectPawn");
            Scribe_Defs.Look(ref activeModule, "activeModule");

        public override void Tick()
        {
            if (!Active)
                return;

            if (effectPawn == null ||
                effectPawn.Dead ||
                Host.Wearer != effectPawn ||
                Module != activeModule ||
                Now >= activeUntil ||
                !effectPawn.health.hediffSet.hediffs.Contains(effect))
            {
                Stop();
                return;
            }

            CloakingModuleExtension settings =
                activeModule?.GetModExtension<CloakingModuleExtension>();

            if (settings == null)
            {
                Stop();
                return;
            }

            int interval = Math.Max(1, settings.energyIntervalTicks);

            // Initialize the timer for saves made before interval payments.
            if (nextEnergyPaymentTick <= 0)
            {
                nextEnergyPaymentTick = Now + interval;
                return;
            }

            if (Now < nextEnergyPaymentTick)
                return;

            float cost = Mathf.Max(0f, settings.energyPerInterval);

            // Disruption does not suspend energy consumption.
            if (cost > 0f)
            {
                IArmorEnergy energy = Host.GetSystem<ArmorEnergySystem>();

                if (energy == null || !energy.TrySpend(cost))
                {
                    Stop();
                    return;
                }
            }

        }

        {
            if (Active && Module != activeModule)
                Stop();
        }

        public override void ModuleRemoved(ThingDef module)
        {
            if (Active && module == activeModule)
                Stop();
        }

        private string BlockedReason()
        {
            CloakingModuleExtension settings = Settings;
            Pawn pawn = Host.Wearer;

            if (!ModsConfig.RoyaltyActive)
            {
                return ModuleUtility.T(
                    "MA_CloakRequiresRoyalty",
                    "Requires Royalty.");
            }

            if (pawn == null || !pawn.Spawned || pawn.Dead || pawn.Downed)
            {
                return ModuleUtility.T(
                    "MA_CloakUnavailable",
                    "The wearer must be standing on a map.");
            }

            if (settings?.hediff == null)
            {
                return ModuleUtility.T(
                    "MA_CloakMissingEffect",
                    "Cloaking effect is not configured.");
            }

            if (Now < cooldownUntil)
            {
                return string.Format(
                    ModuleUtility.T(
                        "MA_CloakCooldown",
                        "Cooldown: {0:0.0} seconds"),
                    (cooldownUntil - Now) / 60f);
            }

            // Do not stack armor cloaking with another invisibility effect.
            if (pawn.GetInvisibilityComp() != null)
            {
                return ModuleUtility.T(
                    "MA_CloakOtherEffect",
                    "Another invisibility effect is already active.");
            }

            float cost = Mathf.Max(0f, settings.energyPerInterval);

            // Require enough stored energy for one payment.
            // Activation itself does not consume energy.
            if (cost > 0f)
            {
                IArmorEnergy energy = Host.GetSystem<ArmorEnergySystem>();

                if (energy == null || energy.Energy < cost)
                {
                    return ModuleUtility.T(
                        "MA_CloakNoEnergy",
                        "Not enough stored energy.");
                }
            }

            return null;
        }

            Pawn pawn = Host.Wearer;

                    "Disappears and Invisibility components.");
                return;

            timer.ticksToDisappear = duration;

            effectPawn = pawn;
            activeModule = Module;
            activeUntil = Now + duration;

            // The first payment occurs after one full interval.
            nextEnergyPaymentTick =

        }

            }
        }

            HashSet<MethodInfo> patched = new HashSet<MethodInfo>();

            foreach (Type type in GenTypes.AllTypes)
            {
                if (!typeof(Verb).IsAssignableFrom(type))
                    continue;

                MethodInfo method = type.GetMethod(
                    "TryCastShot",

                    BindingFlags.DeclaredOnly,

                {
                    harmony.Patch(
                        method,
                        postfix: new HarmonyMethod(postfix));

                {
                    Log.Warning(
                        "[ModularArmor] Cannot patch cloaking attack hook: " +
                        type.FullName + "\n" + exception);
                }
            }
        }

            Verb __instance,

                return;
            }

            if (pawn?.apparel == null)
                return;

            foreach (Apparel apparel in pawn.apparel.WornApparel)
            {
                apparel.TryGetComp<CompModuleSlots>()
                    ?.GetSystem<ArmorCloakingSystem>()
                    ?.NotifySuccessfulAttack();
            }
        }

using System;
using System.Collections.Generic;
using System.Linq;
using RimWorld;
using UnityEngine;
using Verse;

namespace ModularArmor
{
    public class ArmorEnergySystem : ArmorSystem, IArmorEnergy
    {
        private float energy;
        private bool runningPowerAvailable;
        private int lastEnergyTick = -1;
        private float receivedPer60;
        private float sentPer60;

        public float Energy => energy;

        public float Capacity =>
            Host.Systems.Sum(s =>
                Mathf.Max(0f, s.CapacityContribution));

        public bool RunningPowerAvailable =>
            runningPowerAvailable;

        public float GenerationPer60Total =>
            Host.Systems.Sum(s =>
                Mathf.Max(0f, s.GenerationPer60));

        private bool HasSource =>
            Host.IsCuirass &&
            Host.Systems.Any(s => s.IsEnergySource);

        public bool TrySpend(float amount)
        {
            if (float.IsNaN(amount) ||
                float.IsInfinity(amount) ||
                amount < 0f ||
                energy + 0.000001f < amount)
                return false;

            energy = Mathf.Max(0f, energy - amount);

            if (energy <= 0f)
            {
                foreach (ArmorSystem system in Host.Systems)
                    system.EnergyEmpty();
            }

            Host.NotifyStatsChanged();
            return true;
        }

        public override void ExposeData()
        {
            Scribe_Values.Look(
                ref energy,
                "maEnergy",
                0f);
        }

        public override void AfterLoad()
        {
            energy = Mathf.Clamp(energy, 0f, Capacity);
            runningPowerAvailable = false;
            lastEnergyTick = -1;
        }

        public override void ModulesChanged()
        {
            energy = Mathf.Clamp(energy, 0f, Capacity);

            if (Capacity <= 0f)
            {
                runningPowerAvailable = false;

                foreach (ArmorSystem system in Host.Systems)
                    system.PowerLost();
            }
        }

        public override void Tick()
        {
            if (Host.Wearer == null)
            {
                energy = Mathf.Clamp(energy, 0f, Capacity);
                runningPowerAvailable = false;
                receivedPer60 = 0f;
                sentPer60 = 0f;

                foreach (ArmorSystem system in Host.Systems)
                    system.PowerLost();

                return;
            }

            ArmorEnergySystem charger =
                Host.GetWornComponents()
                    .Select(c => c.GetSystem<ArmorEnergySystem>())
                    .FirstOrDefault(e =>
                        e != null && e.HasSource);

            if (charger != null)
                charger.TickWearerEnergy();
            else
                ProcessLocalEnergy(0f);
        }

        private void TickWearerEnergy()
        {
            if (lastEnergyTick == Find.TickManager.TicksGame)
                return;

            List<CompModuleSlots> worn =
                Host.GetWornComponents();

            float surplus = ProcessLocalEnergy(0f);
            float sent = 0f;

            foreach (CompModuleSlots comp in worn)
            {
                if (comp == Host || comp.IsCuirass)
                    continue;

                ArmorEnergySystem target =
                    comp.GetSystem<ArmorEnergySystem>();

                if (target == null)
                    continue;

                float offered = surplus;
                surplus = target.ProcessLocalEnergy(offered);

                sent += Mathf.Max(0f, offered - surplus);
            }

            sentPer60 = sent * 60f;

            foreach (CompModuleSlots comp in worn)
            {
                if (comp != Host && comp.IsCuirass)
                {
                    comp.GetSystem<ArmorEnergySystem>()
                        ?.ProcessLocalEnergy(0f);
                }
            }
        }

        private float ProcessLocalEnergy(float incoming)
        {
            int tick = Find.TickManager.TicksGame;

            if (lastEnergyTick == tick)
                return Mathf.Max(0f, incoming);

            lastEnergyTick = tick;
            receivedPer60 = 0f;
            sentPer60 = 0f;

            incoming = Mathf.Max(0f, incoming);
            energy = Mathf.Clamp(energy, 0f, Capacity);

            if (Host.Wearer == null || Capacity <= 0f)
            {
                runningPowerAvailable = false;

                foreach (ArmorSystem system in Host.Systems)
                    system.PowerLost();

                return incoming;
            }

            float generated = Host.Systems.Sum(s =>
                Mathf.Max(0f, s.GenerateForTick()));

            var budget = new ArmorEnergyBudget(
                energy + generated + incoming,
                Capacity);

            runningPowerAvailable = true;

            foreach (ArmorSystem system in Host.Systems)
                system.BeforeRunningEnergy(budget);

            float cost = Host.RunningConsumption / 60f;

            if (!budget.PayRunningCost(cost))
            {
                runningPowerAvailable = false;

                foreach (ArmorSystem system in Host.Systems)
                    system.PowerLost();
            }
            else
            {
                runningPowerAvailable =
                    cost > 0f || budget.Available > 0f;

                if (runningPowerAvailable)
                {
                    foreach (ArmorSystem system in Host.Systems)
                        system.AfterRunningEnergy(budget);
                }

                if (budget.Available <= 0f)
                {
                    foreach (ArmorSystem system in Host.Systems)
                        system.EnergyEmpty();
                }
            }

            energy = Mathf.Clamp(
                budget.Available,
                0f,
                Capacity);

            float overflow = Mathf.Max(
                0f,
                budget.Available - Capacity);

            receivedPer60 = Mathf.Clamp(
                incoming - overflow,
                0f,
                incoming) * 60f;

            return overflow;
        }

        public string Status =>
            ModuleUtility.T("MA_LocalEnergy", "Local energy: ") +
            energy.ToString("0.0") + "/" +
            Capacity.ToString("0.#") + "\n" +
            ModuleUtility.T("MA_LocalGeneration", "Generation: ") +
            GenerationPer60Total.ToString("0.##") + " | " +
            ModuleUtility.T("MA_LocalDrain", "Running drain: ") +
            Host.RunningConsumption.ToString("0.##") +
            ModuleUtility.T("MA_Per60", " / 60 ticks") +
            "\n" +
            (Host.IsCuirass
                ? ModuleUtility.T(
                    "MA_SentToHelmet",
                    "Sent to helmet: ") +
                  sentPer60.ToString("0.##")
                : ModuleUtility.T(
                    "MA_ReceivedFromCuirass",
                    "Received from cuirass: ") +
                  receivedPer60.ToString("0.##"));

        public override void DrawSettingsHeader(
            Rect area,
            ref float y)
        {
            if (Capacity <= 0f)
                return;

            Widgets.Label(
                new Rect(0f, y, area.width, 75f),
                Status);

            y += 80f;
        }

        public override IEnumerable<Gizmo> GetWornGizmos()
        {
            if (Capacity <= 0f ||
                Find.Selector.SingleSelectedThing != Host.Wearer)
                yield break;

            var bars = new List<ArmorStatusBar>
            {
                new ArmorStatusBar
                {
                    Label = ModuleUtility.T("MA_EnergyBar", "Energy"),
                    Value = energy,
                    Maximum = Capacity,
                    Color = new Color(0.92f, 0.75f, 0.22f)
                }
            };

            ArmorStatusBar secondary =
                Host.Systems
                    .Select(s => s.SecondaryEnergyBar)
                    .FirstOrDefault(b => b != null);

            if (secondary != null)
                bars.Add(secondary);

            yield return new Gizmo_ArmorStatus(
                Host.IsCuirass
                    ? ModuleUtility.T(
                        "MA_CuirassEnergyPanel",
                        "Cuirass — energy")
                    : ModuleUtility.T(
                        "MA_HelmetEnergyPanel",
                        "Helmet — energy"),
                Status,
                bars,
                null,
                null,
                Host.IsCuirass ? -100f : -98f);
        }
    }
}

            Host.InstalledModules.FirstOrDefault(d =>

            overloadTicks = Math.Max(0, overloadTicks);
            cooldownTicks = Math.Max(0, cooldownTicks);
            criticalWearTicks = Math.Max(0, criticalWearTicks);

            if (overloadTicks <= 0)
            {
                overloadDef = null;
                criticalMode = false;
                criticalWearTicks = 0;
                return;
            }

            ModuleExtension info = ModuleUtility.Info(overloadDef);

            bool valid = overloadDef != null &&
                info != null &&
                Host.InstalledModules.Contains(overloadDef);

            if (valid)
            {
                valid = criticalMode
                    ? info.criticalOverload
                    : info.overload;
            }

            if (!valid)
            {
                Stop();
                return;
            }

            if (criticalMode)
            {
                int interval = Math.Max(
                    1,
                    info.criticalArmorDamageInterval);

                criticalWearTicks = Math.Min(
                    criticalWearTicks,
                    interval - 1);
            }
            else
            {
                criticalWearTicks = 0;
            }

        {
            if (overloadTicks <= 0)
                return;

            ModuleExtension info = ModuleUtility.Info(overloadDef);

            int cooldown = criticalMode
                ? info?.criticalOverloadCooldown ?? 7500
                : info?.overloadCooldown ?? 7500;

            cooldownTicks = Math.Max(
                cooldownTicks,
                Math.Max(0, cooldown));

            overloadTicks = 0;
            overloadDef = null;
            criticalMode = false;
            criticalWearTicks = 0;

            if (criticalMode &&

                Stop();
                return;

            if (overloadTicks == 1)
                Stop();

                overloadTicks--;

        private static float Finite(float value, float fallback)
        {
            return float.IsNaN(value) || float.IsInfinity(value)
                ? fallback
                : value;
        }

        private static float NonNegative(float value)
        {
            return Mathf.Max(0f, Finite(value, 0f));
        }

        private static float Multiply(float a, float b)
        {
            return (float)Math.Min(
                float.MaxValue,
                (double)NonNegative(a) * NonNegative(b));
        }

        private static float ResolveSlope(float directional, float legacy)
        {
            return NonNegative(Finite(directional, -1f) >= 0f
                ? directional : legacy);
        }

        private sealed class RangeModel
        {
            public float BaseMin;
            public float BaseMax;
            public float BaseCoefficientBelow;
            public float BaseCoefficientAbove;
            public float Min;
            public float Max;
            public List<ShieldSettingRange> Ranges;

            public float Multiplier(float value)
            {
                value = Mathf.Clamp(Finite(value, 1f), Min, Max);

                if (value == 1f)
                    return 1f;

                float low = Mathf.Min(1f, value);
                float high = Mathf.Max(1f, value);

                var cuts = new List<float>
                {
                    low, high, 1f, BaseMin, BaseMax
                };

                foreach (ShieldSettingRange range in Ranges)
                {
                    cuts.Add(range.min);
                    cuts.Add(range.max);
                }

                cuts = cuts
                    .Where(x => x >= low && x <= high)
                    .Distinct()
                    .OrderBy(x => x)
                    .ToList();

                double integral = 0d;

                for (int i = 0; i < cuts.Count - 1; i++)
                {
                    float a = cuts[i];
                    float b = cuts[i + 1];
                    float midpoint = a + (b - a) * 0.5f;

                    bool below = midpoint < 1f;
                    float baseSlope = below
                        ? BaseCoefficientBelow : BaseCoefficientAbove;

                    float slope =
                        midpoint >= BaseMin && midpoint <= BaseMax
                            ? baseSlope
                            : Ranges
                                .Where(r =>
                                    midpoint >= r.min &&
                                    midpoint <= r.max)
                                .Select(r => ResolveSlope(
                                    below ? r.coefficientBelow : r.coefficientAbove,
                                    r.coefficient))
                                .DefaultIfEmpty(baseSlope)
                                .Min();

                    integral += (double)(b - a) * slope;
                }

                double result =
                    1d + (value < 1f ? -integral : integral);

                return (float)Math.Min(
                    float.MaxValue, Math.Max(0d, result));
            }
        }

        private RangeModel BuildRange(string setting)
        {
            ModuleExtension generator = Shield;

            float minimum = 0.5f;
            float maximum = 1.5f;
            float coefficient = 1f;
            float coefficientBelow = -1f;
            float coefficientAbove = -1f;

            if (generator != null)
            {
                switch (setting)
                {
                    case "Capacity":
                        minimum = generator.shieldCapacityMin;
                        maximum = generator.shieldCapacityMax;
                        coefficient =
                            generator.shieldCapacityCostCoefficient;
                        coefficientBelow = generator.shieldCapacityCostCoefficientBelow;
                        coefficientAbove = generator.shieldCapacityCostCoefficientAbove;
                        break;

                    case "Protection":
                        minimum = generator.shieldProtectionMin;
                        maximum = generator.shieldProtectionMax;
                        coefficient =
                            generator.shieldDamageCostCoefficient;
                        coefficientBelow = generator.shieldProtectionCostCoefficientBelow;
                        coefficientAbove = generator.shieldProtectionCostCoefficientAbove;
                        break;

                    case "Recharge":
                        minimum = generator.shieldRechargeMin;
                        maximum = generator.shieldRechargeMax;
                        coefficient =
                            generator.shieldRechargeCostCoefficient;
                        coefficientBelow = generator.shieldRechargeCostCoefficientBelow;
                        coefficientAbove = generator.shieldRechargeCostCoefficientAbove;
                        break;
                }
            }

            minimum = Mathf.Clamp(
                Finite(minimum, 0.5f), 0.01f, 1f);
            maximum = Mathf.Max(1f, Finite(maximum, 1.5f));

            var ranges = new List<ShieldSettingRange>();

            foreach (ThingDef module in Host.InstalledModules)
            {
                ModuleExtension extension = ModuleUtility.Info(module);

                if (extension == null ||
                    extension.shield ||
                    extension.shieldSettingRanges == null)
                    continue;

                if (extension.active &&
                    !Host.IsModuleEffective(module))
                    continue;

                foreach (ShieldSettingRange range
                    in extension.shieldSettingRanges)
                {
                    if (range == null ||
                        !string.Equals(
                            range.setting,
                            setting,
                            StringComparison.OrdinalIgnoreCase))
                        continue;

                    float a = Mathf.Max(
                        0.01f, Finite(range.min, 1f));
                    float b = Mathf.Max(
                        0.01f, Finite(range.max, 1f));

                    if (b <= a)
                        continue;

                    ranges.Add(new ShieldSettingRange
                    {
                        setting = setting,
                        min = a,
                        max = b,
                        coefficient = NonNegative(range.coefficient),
                        coefficientBelow = ResolveSlope(
                            range.coefficientBelow, range.coefficient),
                        coefficientAbove = ResolveSlope(
                            range.coefficientAbove, range.coefficient)
                    });
                }
            }

            float availableMin = minimum;
            float availableMax = maximum;
            bool changed;

            do
            {
                changed = false;

                foreach (ShieldSettingRange range in ranges)
                {
                    if (range.min > availableMax + Epsilon ||
                        range.max < availableMin - Epsilon)
                        continue;

                    float nextMin = Mathf.Min(availableMin, range.min);
                    float nextMax = Mathf.Max(availableMax, range.max);

                    if (nextMin != availableMin ||
                        nextMax != availableMax)
                    {
                        availableMin = nextMin;
                        availableMax = nextMax;
                        changed = true;
                    }
                }
            }
            while (changed);

            return new RangeModel
            {
                BaseMin = minimum,
                BaseMax = maximum,
                BaseCoefficientBelow = ResolveSlope(coefficientBelow, coefficient),
                BaseCoefficientAbove = ResolveSlope(coefficientAbove, coefficient),
                Min = availableMin,
                Max = availableMax,
                Ranges = ranges
            };
        }

        public float CostMultiplierFor(
            float capacity,
            float protection,
            float recharge)
        {
            // Capacity affects restart only; protection affects repair only.
            // Keep the capacity argument for compatibility with existing callers.
            return Multiply(
                BuildRange("Protection").Multiplier(protection),
                SharedRechargeCostMultiplier(recharge));
        }

        public override void BeforeRunningEnergy(
            ArmorEnergyBudget budget)
        {

            SanitizeState();

            if (!Enabled ||

                return;

            restartPending = false;
            points = 0f;

            Pawn pawn = Host.Wearer;

            if (pawn == null || !pawn.Spawned)
                return;

            SoundDefOf.EnergyShield_Reset.PlayOneShot(

        public override void AfterRunningEnergy(
            ArmorEnergyBudget budget)
        {
            if (!Ready || FullyRepaired)
                return;

            float missing = Mathf.Max(0f, Maximum - points);
            float wanted = Mathf.Min(
                missing, RegenerationPer60 / 60f);

            if (wanted <= 0f)
                return;

            float price = RepairPricePerPoint;
            float restored = wanted;

            if (price > 0f)
            {
                // Repair only the amount affordable this tick.
                restored = Mathf.Min(
                    wanted, NonNegative(budget.Available) / price);
            }

            if (restored <= 0f)
                return;

            float target = Mathf.Min(Maximum, points + restored);
            float actual = target - points;

            // Do not charge for increments lost to float precision.
            if (actual <= 0f)
                return;

            float payment = Multiply(actual, price);

            if (!budget.TrySpend(payment))
                return;

            points = target;
            lastRepairEnergy = payment;
            lastRepairTick = Find.TickManager.TicksGame;
        }

        private void BreakShield()
        {
            points = 0f;
            resetTicks = Math.Max(
                0, Shield?.shieldResetTicks ?? 3200);
            restartPending = resetTicks == 0;
            lastRepairEnergy = 0f;
            lastRepairTick = -9999;

            Pawn pawn = Host.Wearer;

            if (pawn == null || !pawn.Spawned)
                return;

            EffecterDefOf.Shield_Break.SpawnAttached(
                pawn, pawn.Map, 1.2f);

            FleckMaker.Static(
                pawn.TrueCenter(), pawn.Map,
                FleckDefOf.ExplosionFlash, 12f);

            for (int i = 0; i < 6; i++)
                FleckMaker.ThrowDustPuff(
                    pawn.TrueCenter(), pawn.Map,
                    Rand.Range(0.8f, 1.2f));
        }

        private void Absorb(DamageInfo damage)
        {
            Pawn pawn = Host.Wearer;

            if (pawn == null || !pawn.Spawned)
                return;

            SoundDefOf.EnergyShield_AbsorbDamage.PlayOneShot(
                new TargetInfo(pawn.Position, pawn.Map));

            impactDirection =
                Vector3Utility.HorizontalVectorFromAngle(damage.Angle);

            Vector3 location = pawn.TrueCenter() +
                impactDirection.RotatedBy(180f) * 0.5f;

            float size = Mathf.Min(10f, 2f + damage.Amount / 10f);

            FleckMaker.Static(
                location, pawn.Map,
                FleckDefOf.ExplosionFlash, size);

            for (int i = 0; i < (int)size; i++)
                FleckMaker.ThrowDustPuff(
                    location, pawn.Map, Rand.Range(0.8f, 1.2f));

            lastImpactTick = Find.TickManager.TicksGame;
        }

        public override void BeforeDamage(
            ref DamageInfo damage,
            ref bool absorbed)
        {
            if (absorbed || !Ready || Host.Wearer == null)
                return;

            if (damage.Def == DamageDefOf.EMP)
            {
                BreakShield();
                return;
            }

            if (points <= 0f ||
                damage.Def.ignoreShields ||
                (!damage.Def.isRanged && !damage.Def.isExplosive))
                return;

            absorbed = true;

            points = Mathf.Max(
                0f,
                points - Multiply(damage.Amount, LossPerDamage));

            if (points <= 0f)
                BreakShield();
            else
                Absorb(damage);
        }

using System;
using System.Collections.Generic;
using UnityEngine;
using Verse;

namespace ModularArmor
{
    public class ArmorStatusBar
    {
        public string Label;
        public float Value;
        public float Maximum;
        public Color Color = Color.white;
    }

    public class Gizmo_ArmorStatus : Gizmo
    {
        private readonly string title;
        private readonly string tooltip;
        private readonly List<ArmorStatusBar> bars;
        private readonly string buttonLabel;
        private readonly Action buttonAction;

        public Gizmo_ArmorStatus(
            string title,
            string tooltip,
            List<ArmorStatusBar> bars,
            string buttonLabel,
            Action buttonAction,
            float order)
        {
            this.title = title;
            this.tooltip = tooltip;
            this.bars = bars;
            this.buttonLabel = buttonLabel;
            this.buttonAction = buttonAction;
            Order = order;
        }

        public override float GetWidth(float maxWidth) => 230f;

        public override GizmoResult GizmoOnGUI(
            Vector2 topLeft,
            float maxWidth,
            GizmoRenderParms parms)
        {
            Rect rect = new Rect(
                topLeft.x, topLeft.y, GetWidth(maxWidth), 75f);

            Widgets.DrawWindowBackground(rect);

            GameFont oldFont = Text.Font;
            TextAnchor oldAnchor = Text.Anchor;
            Color oldColor = GUI.color;

            Text.Font = GameFont.Tiny;
            Text.Anchor = TextAnchor.UpperLeft;
            GUI.color = Color.white;

            Widgets.Label(
                new Rect(rect.x + 6f, rect.y + 2f,
                    rect.width - 12f, 17f), title);

            int count = Math.Min(2, bars?.Count ?? 0);

            if (buttonAction != null)
                count = Math.Min(1, count);

            for (int i = 0; i < count; i++)
            {
                ArmorStatusBar bar = bars[i];

                Rect row = new Rect(
                    rect.x + 6f, rect.y + 21f + i * 26f,
                    rect.width - 12f, 20f);

                float fraction = bar.Maximum > 0f
                    ? Mathf.Clamp01(bar.Value / bar.Maximum) : 0f;

                Widgets.DrawBoxSolid(
                    row, new Color(0.12f, 0.12f, 0.12f));

                if (fraction > 0f)
                    Widgets.DrawBoxSolid(
                        new Rect(row.x, row.y,
                            row.width * fraction, row.height),
                        bar.Color);

                Text.Anchor = TextAnchor.MiddleCenter;

                Widgets.Label(
                    row, bar.Label + ": " +
                    bar.Value.ToString("0.0") + "/" +
                    bar.Maximum.ToString("0"));
            }

            if (buttonAction != null)
            {
                Rect button = new Rect(
                    rect.x + 6f, rect.y + 47f,
                    rect.width - 12f, 20f);

                if (Widgets.ButtonText(button, buttonLabel))
                    buttonAction();
            }

            TooltipHandler.TipRegion(rect, tooltip ?? "");

            Text.Font = oldFont;
            Text.Anchor = oldAnchor;
            GUI.color = oldColor;

            return new GizmoResult(GizmoState.Clear);
        }
    }
}
